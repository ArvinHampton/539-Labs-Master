\subsection{Improper automorphs and the orbit of the corner}\label{sec:improper}
Proposition~\ref{prop:transport} is stated for $g\in\mathrm{GL}_2(\Z)$ with $g^{\mathsf T}Ag=A$ and its proof does not use $\det g$; it therefore applies verbatim to improper automorphs, with
$\Gamma_g$ defined by the same reduction rules. From the defining relation, $\Gamma_{g_1g_2}=\Gamma_{g_2}\Gamma_{g_1}$.
\begin{lemma}\label{lem:parity-improper}
For every unimodular cone $C=\R_{\ge0}u_1+\R_{\ge0}u_2$, $\mathrm{Cor}_{-C}=\mathrm{Cor}_C$.
\end{lemma}
\begin{proof}
In the basis $(-u_1,-u_2)$ the coordinates of a point are $(-a,-b)$; $B(-u_1,-u_2)=B(u_1,u_2)$ and $Q(-u_i)=Q(u_i)$. Since $\bar B_1$ is odd and vanishes on $\Z$ and $B_2$ is even, each
summand of the corner formula is unchanged, with the same phases.
\end{proof}
\begin{proposition}[exact, computer-assisted]\label{prop:improper}
Let $h=\begin{psmallmatrix}1&7\\0&-1\end{psmallmatrix}=-\sigma$ and $K=\R_{\ge0}^2$. Then $\Gamma_h=\Gamma_{-I}\neq I$, $\Gamma_h^2=I$, $\Gamma_hM=M\Gamma_h$, $\Gamma_h\Gamma_\gamma=\Gamma_\gamma\Gamma_h$, and
$\Gamma_h A_2^{\rm corner}=\mathrm{Cor}_{hK}=A_2^{\rm corner}$.
\end{proposition}
\begin{proof}
The matrix identities are finite checks on permutations and exponents modulo $9800$; the commutator with $M$ is checked on exponent multisets as in Proposition~\ref{prop:gammainv}
(script \path{BRIDGE_C_35/order27/scripts/o27.py}). Since $hK=-(\sigma K)$, Lemma~\ref{lem:parity-improper} and Corollary~\ref{cor:sigma} give $\mathrm{Cor}_{hK}=\mathrm{Cor}_{\sigma K}=A_2^{\rm corner}$, and
Proposition~\ref{prop:transport} identifies this with $\Gamma_hA_2^{\rm corner}$.
\end{proof}
\begin{corollary}\label{cor:orbit}
For every $w$ in the group generated by $\gamma$ and $h$, $\mathrm{Cor}_{wK}\in\{A_2^{\rm corner},\,\Gamma_\gamma A_2^{\rm corner}\}$. Consequently every linear combination of corners over this
orbit, including all group averages and all character projections, lies in $\mathrm{span}\{Y_+,Y_-\}$ of Corollary~\ref{cor:split}.
\end{corollary}
\begin{proof}
$\mathrm{Cor}_{wK}=\Gamma_{w^{-1}}A_2^{\rm corner}$, and $\Gamma_{w^{-1}}$ is a word in $\Gamma_\gamma,\Gamma_h$. These commute, $\Gamma_\gamma^2=\Gamma_h^2=I$, and $\Gamma_h$ fixes $A_2^{\rm corner}$.
\end{proof}
\begin{catbremark}
On the $216$-point grid the ten cones $\gamma^ah^bK$, $|a|\le2$, $b\in\{0,1\}$, and $-K$ agree with the orbit of Corollary~\ref{cor:orbit} to $10^{-14}$; the four character projectors
have ranks $36,24,18,27$ (Appendix~\ref{app:catB}, item~\ref{catB:o27}).
\end{catbremark}
